// ══════════════════════════════════════════════════════════════════════
//  Platizio – Google Apps Script backend
// ══════════════════════════════════════════════════════════════════════
//
//  Handles THREE payload shapes, all POSTed to the same Web App URL:
//    • formType "Net Worth Certificate"  → "Net Worth Certificate" tab
//    • formType "Accreditation"          → "Accreditation" tab
//    • no formType (legacy index.html)   → the first/active sheet tab
//
//  The two new tabs are created and formatted automatically on first
//  submission, so you don't strictly need to run setupSheetHeaders() —
//  but you can run it once to pre-create and format all tabs.
//
//  SETUP (do this once before deploying):
//  1. Create a Google Sheet and copy its ID into SHEET_ID below.
//     (The ID is the long string in the Sheet URL between /d/ and /edit)
//  2. Create a Google Drive folder for uploaded documents and copy
//     its ID into FOLDER_ID below.
//     (The ID is the long string at the end of the folder URL)
//  3. (Optional) Run setupSheetHeaders() once to pre-create the tabs.
//  4. Deploy as a Web App:
//       Extensions → Apps Script → Deploy → New deployment
//       Type: Web App · Execute as: Me · Who has access: Anyone
//  5. The Web App URL is already wired into the site (lib/site.ts) and
//     index.html → WEB_APP_URL. Re-deploy a NEW version after any edit.
// ══════════════════════════════════════════════════════════════════════

const SHEET_ID  = 'YOUR_SHEET_ID_HERE';   // ← replace with your Google Sheet ID
const FOLDER_ID = 'YOUR_FOLDER_ID_HERE';  // ← replace with your Drive folder ID

const NW_TAB  = 'Net Worth Certificate';
const ACC_TAB = 'Accreditation';

const BRAND = '#7A2000';


// ══════════════════════════════════════════════════════════════════════
//  Entry point — route by formType
// ══════════════════════════════════════════════════════════════════════

function doPost(e) {
  try {
    const data     = JSON.parse(e.postData.contents);
    const ss       = SpreadsheetApp.openById(SHEET_ID);
    const folder   = DriveApp.getFolderById(FOLDER_ID);
    const formType = (data.formType || '').toLowerCase();

    if (formType.indexOf('net') !== -1 && formType.indexOf('worth') !== -1) {
      handleNetWorth(data, ss, folder);
    } else if (formType.indexOf('accredit') !== -1) {
      handleAccreditation(data, ss, folder);
    } else {
      handleLegacy(data, ss, folder);   // index.html single-flow payload
    }

    return jsonOut({ status: 'success' });

  } catch (error) {
    return jsonOut({ status: 'error', message: error.toString() });
  }
}


// ══════════════════════════════════════════════════════════════════════
//  Shared helpers
// ══════════════════════════════════════════════════════════════════════

function jsonOut(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// Build file uploaders bound to one applicant subfolder.
function makeUploaders(subFolder) {

  // Single base64-encoded file → shareable Drive link ('' if absent)
  function uploadFile(fileObj, label) {
    if (!fileObj || !fileObj.data) return '';
    const base64Part = fileObj.data.indexOf(',') !== -1
      ? fileObj.data.split(',')[1]
      : fileObj.data;
    const decoded = Utilities.base64Decode(base64Part);
    const blob    = Utilities.newBlob(decoded, fileObj.type || 'application/octet-stream', label + '_' + fileObj.name);
    const file    = subFolder.createFile(blob);
    try {
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (sharingErr) {
      Logger.log('Sharing restricted for ' + label + ': ' + sharingErr.toString());
    }
    return file.getUrl();
  }

  // Array of base64-encoded files → comma-separated Drive links
  function uploadFiles(fileArr, label) {
    if (!fileArr || !fileArr.length) return '';
    return fileArr
      .filter(f => f && f.data)
      .map((f, i) => uploadFile(f, label + '_' + (i + 1)))
      .join(', ');
  }

  return { uploadFile: uploadFile, uploadFiles: uploadFiles };
}

// One applicant subfolder per submission, prefixed by form type.
function makeSubFolder(folder, data, prefix) {
  const name = (data.fullName || 'Applicant').trim();
  const pan  = (data.panCard  || 'NOPAN').trim();
  const ts   = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyyMMdd_HHmm');
  return folder.createFolder(prefix + '_' + name + '_' + pan + '_' + ts);
}

// Compact an address object into a single sheet cell
function formatAddress(a) {
  a = a || {};
  return [a.line1, a.line2, a.city, a.country, a.pincode]
    .filter(p => p && String(p).trim())
    .join(', ');
}

function residencyLabel(data) {
  return (data.applicationType || '').toLowerCase() === 'nri' ? 'NRI' : 'Indian Resident';
}

function validityLabel(data) {
  const v = (data.certValidity || '').toLowerCase().replace(/[^a-z0-9]/g, '');
  return (v === '3year' || v === 'threeyear') ? '3-Year' : '2-Year';
}

function pathLabel(data) {
  const p = (data.eligibilityPath || '').toLowerCase().replace(/\s+/g, '');
  return p === 'networth' ? 'Net Worth'
       : p === 'hybrid'   ? 'Hybrid'
       : p === 'income'   ? 'Income'
       : (data.eligibilityPath || '');
}

// Get a tab by name, creating + formatting its header row if it's new/empty.
function ensureSheet(ss, name, headers) {
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  if (sheet.getLastRow() === 0) {
    const range = sheet.getRange(1, 1, 1, headers.length);
    range.setValues([headers]);
    range.setFontWeight('bold');
    range.setBackground(BRAND);
    range.setFontColor('#ffffff');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

// Shared applicant columns (identical for both new forms) ───────────────
function applicantHeaders() {
  return [
    'Timestamp', 'Form Type', 'Application Type', 'Account Type',
    'Full Name', 'Organization', 'Email Address', 'Phone Number',
    'PAN Number', 'Aadhaar Number', 'Current Address', 'Permanent Address',
    'Spouse – Full Name', 'Spouse – Organization', 'Spouse – Email',
    'Spouse – Phone', 'Spouse – PAN', 'Spouse – Aadhaar',
    'Spouse – Current Address', 'Spouse – Permanent Address',
  ];
}

function applicantCells(data) {
  const sp = data.spouse || {};
  return [
    new Date(),
    data.formType || '',
    residencyLabel(data),
    data.accountType   || '',
    data.fullName      || '',
    data.organization  || '',
    data.emailAddress  || '',
    data.phoneNumber   || '',
    data.panCard       || '',
    data.aadhaarNumber || '',
    formatAddress(data.currentAddress),
    formatAddress(data.permanentAddress),
    sp.fullName      || '',
    sp.organization  || '',
    sp.emailAddress  || '',
    sp.phoneNumber   || '',
    sp.panCard       || '',
    sp.aadhaarNumber || '',
    formatAddress(sp.currentAddress),
    formatAddress(sp.permanentAddress),
  ];
}


// ══════════════════════════════════════════════════════════════════════
//  Net Worth Certificate form
// ══════════════════════════════════════════════════════════════════════

const NW_HEADERS = applicantHeaders().concat([
  'Certificate Validity',                          // U
  'ITR – Latest Year (Drive Link)',                // V
  'ITR – Previous Year (Drive Link) [3-Year only]',// W
  'CAS Statement (Drive Link)',                    // X
  'PMS / AIF / SIF Statements (Drive Links)',      // Y
  'Bank Statements (Drive Links)',                 // Z
  'PF / EPF Statements (Drive Links)',             // AA
  'International Equities (Drive Links)',           // AB
  'Real Estate – Ownership Proof (Drive Links)',   // AC
  'Real Estate – Ready Reckoner Value (Drive Links)', // AD
  'Gold Valuation Certificate (Drive Link)',       // AE
  'Jewellery Valuation Certificate (Drive Link)',  // AF
  'Loan Statements / Balance Sheet (Drive Links)', // AG
]);

function handleNetWorth(data, ss, folder) {
  const name = (data.fullName || 'Applicant').trim();
  const subFolder = makeSubFolder(folder, data, 'NWC');
  const up = makeUploaders(subFolder);
  const docs = data.documents || {};

  const row = applicantCells(data).concat([
    validityLabel(data),
    up.uploadFile(docs.itr,                  name + '_ITR_Latest'),
    docs.itrPrev ? up.uploadFile(docs.itrPrev, name + '_ITR_Previous') : '',
    up.uploadFile(docs.cas,                  name + '_CAS'),
    up.uploadFiles(docs.pms,                 name + '_PMS'),
    up.uploadFiles(docs.bank,                name + '_Bank'),
    up.uploadFiles(docs.pf,                  name + '_PF'),
    up.uploadFiles(docs.intl,                name + '_Intl'),
    up.uploadFiles(docs.realEstateProof,     name + '_RE_OwnershipProof'),
    up.uploadFiles(docs.realEstateValue,     name + '_RE_ReadyReckoner'),
    up.uploadFile(docs.gold,                 name + '_Gold'),
    up.uploadFile(docs.jewellery,            name + '_Jewellery'),
    up.uploadFiles(docs.loans,               name + '_Loans'),
  ]);

  ensureSheet(ss, NW_TAB, NW_HEADERS).appendRow(row);
}


// ══════════════════════════════════════════════════════════════════════
//  Accreditation form
// ══════════════════════════════════════════════════════════════════════

// Document requirements depend on the eligibility path:
//   Net Worth → certificate · Income → ITR · Hybrid → both.
// Unused columns are left blank for that submission.
const ACC_HEADERS = applicantHeaders().concat([
  'Eligibility Path',                              // U
  'Certificate Validity',                          // V
  'ITR – Latest Year (Drive Link) [Income/Hybrid]',// W
  'ITR – Previous Year (Drive Link) [3-Year only]',// X
  'Net Worth Certificate (Drive Link) [Net Worth/Hybrid]', // Y
  'PAN Copy (Drive Link)',                         // Z
  'Aadhaar Copy (Drive Link)',                     // AA
  'Marriage Certificate (Drive Link) [Joint]',     // AB
  'Signed Undertakings (Drive Links)',             // AC
]);

function handleAccreditation(data, ss, folder) {
  const name = (data.fullName || 'Applicant').trim();
  const subFolder = makeSubFolder(folder, data, 'ACC');
  const up = makeUploaders(subFolder);
  const docs = data.documents || {};

  const row = applicantCells(data).concat([
    pathLabel(data),
    validityLabel(data),
    up.uploadFile(docs.itr,                  name + '_ITR_Latest'),
    docs.itrPrev ? up.uploadFile(docs.itrPrev, name + '_ITR_Previous') : '',
    up.uploadFile(docs.nwCertificate,        name + '_NetWorth_Certificate'),
    up.uploadFile(docs.panCopy,              name + '_PAN_Copy'),
    up.uploadFile(docs.aadhaarCopy,          name + '_Aadhaar_Copy'),
    docs.marriageCertificate ? up.uploadFile(docs.marriageCertificate, name + '_Marriage_Certificate') : '',
    up.uploadFiles(docs.undertakings,        name + '_Undertakings'),
  ]);

  ensureSheet(ss, ACC_TAB, ACC_HEADERS).appendRow(row);
}

  'Spouse – Phone Number', 'Spouse – PAN Number', 'Spouse – Aadhaar Number',
  'Spouse – Current Address', 'Spouse – Permanent Address',
];

function handleLegacy(data, ss, folder) {
  const name = (data.fullName || 'Applicant').trim();
  const subFolder = makeSubFolder(folder, data, 'LEGACY');
  const up = makeUploaders(subFolder);

  const docs = data.documents || {};
  const path = (data.eligibilityPath || '').toLowerCase().replace(/\s+/g, '');

  const itrUrl     = up.uploadFile(docs.itr,     name + '_ITR_Latest');
  const itrPrevUrl = docs.itrPrev ? up.uploadFile(docs.itrPrev, name + '_ITR_Previous') : '';
  const nwCertUrl  = up.uploadFile(docs.nwCertificate, name + '_NetWorth_Certificate');

  let casUrl = '', pmsUrls = '', bankUrls = '', pfUrls = '', intlUrls = '';
  let realEstateProofUrls = '', realEstateValueUrls = '', goldUrl = '', jewelleryUrl = '';
  let loansUrls = '', realEstateUrls_hy = '', goldUrls_hy = '';

  if (path === 'networth') {
    casUrl              = up.uploadFile(docs.cas,              name + '_CAS');
    pmsUrls             = up.uploadFiles(docs.pms,             name + '_PMS');
    bankUrls            = up.uploadFiles(docs.bank,            name + '_Bank');
    pfUrls              = up.uploadFiles(docs.pf,              name + '_PF');
    intlUrls            = up.uploadFiles(docs.intl,            name + '_Intl');
    realEstateProofUrls = up.uploadFiles(docs.realEstateProof, name + '_RE_OwnershipProof');
    realEstateValueUrls = up.uploadFiles(docs.realEstateValue, name + '_RE_ReadyReckoner');
    goldUrl             = up.uploadFile(docs.gold,             name + '_Gold');
    jewelleryUrl        = up.uploadFile(docs.jewellery,        name + '_Jewellery');
    loansUrls           = up.uploadFiles(docs.loans,           name + '_Loans');
  } else if (path === 'hybrid') {
    casUrl            = up.uploadFile(docs.cas,         name + '_CAS');
    pmsUrls           = up.uploadFiles(docs.pms,        name + '_PMS');
    bankUrls          = up.uploadFiles(docs.bank,       name + '_Bank');
    pfUrls            = up.uploadFiles(docs.pf,         name + '_PF');
    realEstateUrls_hy = up.uploadFiles(docs.realEstate, name + '_RealEstate');
    goldUrls_hy       = up.uploadFiles(docs.gold,       name + '_Gold_Jewellery');
    loansUrls         = up.uploadFiles(docs.loans,      name + '_Loans');
  }
  // Income path: ITR only (already handled above)

  const curr = data.currentAddress   || {};
  const perm = data.permanentAddress || {};
  const spouse = data.spouse || {};

  const appTypeLabel =
    (data.applicationType || '').toLowerCase() === 'nri'
      ? 'Individual (NRI)' : 'Individual (Indian Resident)';

  const sheet = ss.getSheets()[0];   // first/active tab (existing legacy data)
  if (sheet.getLastRow() === 0) ensureSheet(ss, sheet.getName(), LEGACY_HEADERS);

  sheet.appendRow([
    new Date(), appTypeLabel, validityLabel(data),
    data.fullName || '', data.organization || '', data.emailAddress || '',
    data.phoneNumber || '', data.panCard || '',
    curr.line1 || '', curr.line2 || '', curr.city || '', curr.country || '', curr.pincode || '',
    perm.line1 || '', perm.line2 || '', perm.city || '', perm.country || '', perm.pincode || '',
    pathLabel(data),
    itrUrl, itrPrevUrl, casUrl, pmsUrls, bankUrls, pfUrls, intlUrls,
    realEstateProofUrls, realEstateValueUrls, goldUrl, jewelleryUrl,
    loansUrls, realEstateUrls_hy, goldUrls_hy,
    data.accountType || '', data.aadhaarNumber || '', data.certMethod || '', nwCertUrl,
    spouse.fullName || '', spouse.organization || '', spouse.emailAddress || '',
    spouse.phoneNumber || '', spouse.panCard || '', spouse.aadhaarNumber || '',
    formatAddress(spouse.currentAddress), formatAddress(spouse.permanentAddress),
  ]);
}


// ══════════════════════════════════════════════════════════════════════
//  setupSheetHeaders()
//  Optional — run ONCE manually to pre-create and format all three tabs.
//  (Tabs are also auto-created on first submission.)
//
//  How to run: Apps Script editor → select setupSheetHeaders from the
//  function dropdown → click Run.
// ══════════════════════════════════════════════════════════════════════

function setupSheetHeaders() {
  const ss = SpreadsheetApp.openById(SHEET_ID);

  // Legacy tab = first sheet (keeps existing index.html data)
  const legacy = ss.getSheets()[0];
  if (legacy.getLastRow() === 0) ensureSheet(ss, legacy.getName(), LEGACY_HEADERS);

  // New form tabs
  ensureSheet(ss, NW_TAB,  NW_HEADERS);
  ensureSheet(ss, ACC_TAB, ACC_HEADERS);

  // Auto-resize for readability
  [legacy, ss.getSheetByName(NW_TAB), ss.getSheetByName(ACC_TAB)].forEach(s => {
    if (s && s.getLastColumn() > 0) s.autoResizeColumns(1, s.getLastColumn());
  });

  Logger.log('Tabs ready: "' + legacy.getName() + '" (legacy), "' + NW_TAB + '", "' + ACC_TAB + '".');
}
